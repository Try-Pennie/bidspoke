const lendswiftData = steps.ping_lendswift.output.data.data;
const lendswiftResult = lendswiftData.status;
const lendswiftBid = lendswiftData.bid_amount;
const constants = steps.constants.output;
const subAccountUuid = payload.subAccountUuid;

const floor = constants.LendswiftSubAccountMinBids?.[subAccountUuid];

// Shared values used by the patch_applicant_* Salesforce Update nodes.
// (Templates can't do math / ?? / || / dynamic keys, so we compute here.)
const patch = {
  externalPartnerBid: lendswiftBid,
  minBidAmt: (lendswiftData && lendswiftData.min_bid_value != null) ? lendswiftData.min_bid_value : floor,
  redirectUrl: lendswiftData && lendswiftData.redirect_url,
  rejectionReason: lendswiftData && lendswiftData.rejection_reason,
  ts: new Date().toISOString(),
};

// --- unqualified ---
if (!Number(lendswiftBid) || Number(lendswiftBid) <= 0 || lendswiftResult !== "ACCEPTED") {
  const reason = "unqualified_lead";
  return {
    body: { lead_id: payload.leadUuid, status: "rejected", reason },
    ...patch,
    lsRejectionReason: patch.rejectionReason || reason,
  };
}

// --- below threshold ---
if (floor !== undefined && Number(lendswiftBid) < floor) {
  const reason = "bid_below_threshold";
  return {
    body: { lead_id: payload.leadUuid, status: "rejected", reason },
    ...patch,
    lsRejectionReason: patch.rejectionReason || reason,
  };
}

// --- offer (BID) ---
const bidAmount = lendswiftBid * constants.multiplier;
return {
  body: {
    requestId: execution.trace_id,
    offers: [
      {
        name: constants.offerName,
        display_name: constants.offerName,
        rate: bidAmount,
        click_url: constants.pennieClickUrl + `/${steps.create_applicant.output.body.id}`,
        product_type: constants.productType,
        logo_src: constants.logoSrc,
        offer_title: constants.offerTitle,
        offer_details: constants.offerDetails,
      },
    ],
  },
  ...patch,
  bidAmount,
};
